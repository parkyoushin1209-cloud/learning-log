struct Link {
	string value;
	Link* prev;
	Link* succ;
	Link(const string& v, Link* p, Link* s)
		:value{v}, prev{p}, succ{s}{ }
	Link(const string& v)
		:value{v}, prev{nullptr}, succ{nullptr}{ }


};

Link* insert(Link* p, Link* n) { // 링크 삽입
	if (p == nullptr) return n;
	if (n == nullptr)return p;
	n->succ = p;
	if (p->prev) p->prev->succ = n; // p->prev가 존재하면 연결
	n->prev = p->prev;
	p->prev = n;
	return n; // 추가 삽입할 노드 반환
	// n의 앞뒤로만 연결 제대로 해주기 (t->n->p) 이므로 t->succ, p->prev, n->prev/succ만

}

Link* add(Link* p, Link* n) { // n을 p 뒤에 삽입하고 n을 반환
	if (p == nullptr) return n;
	if (n == nullptr) return p;
	n->succ = p->succ;
	n->prev = p;
	if (p->succ != nullptr)
		p->succ->prev = n;
	p->succ = n;
	return n;
}

Link* erase(Link* p) {
	if (p == nullptr) return nullptr;
	if (p->succ) p->succ->prev = p->prev;
	if (p->prev) p->prev->succ = p->succ;
	return p->succ;
}

Link* find(Link* p, const string& s) {
	while (p) {
		if (p->value == s)return p;
		p = p->succ;
	}
	return nullptr;
}

Link* advance(Link* p, int n) {
	if (p == nullptr) return nullptr;
	if (n > 0) {
		while (n--) { // 후미 연산자 => 한 번 실행 후 감소한 뒤 검사
			if (p->succ == nullptr) return nullptr;
			p = p->succ;
		}
	}
	else if (n < 0) {
		while (n++) {
			if (p->prev == nullptr) return nullptr;
			p = p->prev;
		}
	}
	return p;
}

void print_all(Link* p) {
	cout << "{ ";
	while (p) {
		cout << p->value;
		if (p = p->succ) cout << ", ";
	}
	cout << " }";
}

void to_lower(char* s) {
	while (*s) {
		*s = static_cast<unsigned char>(*s);
		s++;
	}
}

char* my_strdup(const char* s) {
	if (s == nullptr) return nullptr;

	const char* l = s;
	int len = 0;
	while (l) {
		len++;
		l++;
	}

	char* p = new char[len + 1];

	for (int i = 0; i < len; i++)
		p[i] = s[i];

	p[len] = '\0';
	return p;
}

const char* findx(const char* s, const char* x) {
	if (!s || !x) return nullptr;

	while (*s) {
		const char* p = s;
		const char* dx = x;

		while (*dx && *p == *dx) {
			p++;
			dx++;
		}

		if (*dx == '\0') return s;  // 찾음
		s++;  // 다음 문자로 이동
	}

	return nullptr;  // 못 찾음
}
