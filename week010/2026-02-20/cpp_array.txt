포인터 관련 주의할 점
-널포인터에 접근하지 말기
-존재하지 않는 배열의 요소에 접근하지 말기(범위 넘기지 말기)
-초기화하지 않은 포인터 접근하지 말기
-유효 범위를 벗어난 객체에 접근하지 말기
-해제된 객체에(삭제된 객체에) 접근하지 말기

bool is_palindrome(const string& s) {
	int first = 0;
	int last = s.length() - 1;
	while (first < last) {
		if (s[first] != s[last]) return false;
		first++;
		last++;
	}
	return true;
}

char* strdup(const char* a) {
	const char* p = a;
	int len = 0;
	while (*p) { len++;  p++; }
	p = a;
	char* result = new char[len+1];

	delete[] p;

	const char* src = a;
	char* dest = result; //*가 붙으면 값 복사임

	for (int i = 0; i < len; i++) {
		*dest = *src;
		dest++;
		src++;
	}
	*dest = '\0';
	return result;
}

int strcmp(const char* s, const char* x) {
	const char* s1 = s;
	const char* s2 = x;
	while (*s1 || *s2) {
		if (*s1 < *s2) return -1;
		else if (*s1 > *s2) return 1;
		else {
			s1++;
			s2++;
		}
	}
	if (*s1 && !*s2) return 1;
	else if (*s2 && !*s1) return -1;
	else return 0;
}

string strcat_dot(const string& s1, const string& s2) {
	string s;
	for (char c : s1)  // 널문자는 순회 안 함
		s += c;
	s += '.';
	for (char c : s2)
		s += c;
	//s += '\0' 불필요 내부적으로 관리함
	return s;
}
