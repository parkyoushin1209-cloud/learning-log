vector v2 = v1은 복사 생성자(대입 아님)

class VECTOR {
	int sz;
	double elem;
public

	VECTOR(int s)
		sz{ s }, elem{ new double[s] }
	{
		for (int i = 0; i  sz; i++)
			elem[i] = 0.0;
	}

	VECTOR(const VECTOR&);

	VECTOR(initializer_listdouble lst)
		sz{ lst.size() }, elem{ new double[lst.size()] }
	{
		copy(lst.begin(), lst.end(), elem);
	}

	VECTOR& operator=(const VECTOR&);

	VECTOR(VECTOR&&);

	VECTOR& operator=(VECTOR&&);

	double& operator[](int n) { return elem[n]; }

	double operator[](int n) const { return elem[n]; }
	~VECTOR()
	{
		delete[] elem;
	}

};
 포인터나 참조를 복사할 경우 얕은 복사 = 소멸자가 두 번 호출되버리면 문제
  복사 생성자, 초기화할 때 이용, 복사할 원본 객체의 참조를 인자로 받음
VECTORVECTOR(const VECTOR& arg)  복사 생성자의 인자 자체가 복사되면 안되므로 const로
	sz{ arg.sz }, elem{ new double[arg.sz] }
{
	copy(arg.elem, arg.elem + sz, elem);
}

VECTOR& VECTORoperator=(const VECTOR& a) {  복사 대입자 초기화 이후 대입할 때
	double p = new double[a.sz];
	copy(a.elem, a.elem + sz, p);
	delete[] elem; 이전 공간 반화 = 메모리 누수 방지
	elem = p;  새로 할당
	sz = a.sz;
	return this; 자기 참조 반환
}
vector의 요소가 많다면 복사가 부다되므로 복사를 보완한느 이동 연산 정의(더 단순하고 효율적임)
VECTORVECTOR(VECTOR&& a)  &&는 우항 참조, 이동 연산자 정의시 사용함, 이동 연산자는 const로 받지 않음(이동 연산의 목적 중 하나는 원본을 수정해 빈 객체로 만드는 것임)
	sz{ a.sz }, elem{ a.elem } 이전 것 그대로 주고
{
	a.sz = 0;
	a.elem = nullptr;  이전 것은 초기화
}

VECTOR& VECTORoperator=(VECTOR&& a)
{
	delete[] elem;
	elem = a.elem;
	sz = a.sz;
	a.elem = nullptr;
	a.sz = 0;
	return this;  자기 참조 반환 this는 자기 자신을 가리키는 포인터 this는 역참조
}

명시적인 생성자
인자를 하나를 전달 받는 생성자는 인자의 타입을 해당 타입으로 변환하는 연산을 정의함
vector v =10;을 vector v(10)으로 해석해버릴 수 있음 = explicit vector(int) 사용(암묵적인 변환 불가능하며 객체 생성 수단으로 사용함)

생성자 소멸자는 여러 경우에 호출될 수 있음 (복사, 객체 생성, new.delete 등 호출할 경우) = 디버깅을 위해 출력을 포함해도 됨

double& operator[](int n) {return elem[n];}
을 정의해 요소에 접근 가능하게 이때 값을 반환하면 쓰기는 불가능 포인터를 이용하면 를 붙여야되므로 번거로움
double operator[](int n){return elem[n];}을 이용해서 const 벡터도 받을 수 있게
