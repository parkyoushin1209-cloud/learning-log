DUT테스트에서 중요한 부분은 내부 DUT 환경과 둘러싼 시스템의 구성임 (환경 또한 랜덤화해서 가능한 최대로 테스트되게 해야 함)

////////random device configuration : 실제 환경에 영향을 받는 DUT 묘사
class Ethcfg; // 4포트 이더넷 스위치 구성 묘사
  rand bit[3:0] in_use; // test 3,2,1,0에서 사용되는 포트
  rand bit[47:0] mac_addr[4]; // max주소(media access control address): 네트워크 장치를 식별하는 고유 하드웨어 주소
  rand bit[3:0] is_100; // 포트 3,2,1,0을 위한 100mb 모드 
rand uint run_for_n_frames; 

  constraint local_unicast{ //unicast 모드면 addr에 변화 줌
    foreach(mac_addr[i])
    mac_addr[i][41:40] = 2'b00;
}

constraint reasonable{
  run_for_n_frames inside {[1:100];}; // 테스트 길이 제한
}
endclass

class Environment;
Ethcfg cfg;
EthGen gen[4];
EthMii drv[4];

function void new();
cfg = new();
endfunction

function void build();
foreach(gen[i]) begin
    gen[i] = new();
    drv[i] = new();
    if(cfg.is_100[i])
    drv[i].set_speed(100);
end
endfunction

function void gen_cfg();
'SV_RAND_CHECK(cfg.randomize());
endfunction

task run();
foreach(gen[i])
if(cfg.in_use[i]) begin
    gen[i].run();
    //......
end
endtask

task wrap_up();
//Not currently used
endtask
endclass

program automatic test;
Environment env;
initial begin
env = new(); // 환경 객체 구성
env.gen_cfg(); // 랜덤 구성 생성
//env.cfg.in_use = '1; -> 코너 케이스 도달을 위해 덮어쓰기 가능
env.build(); //테스트벤치 환경 구축
env.run(); // 테스트 실행
env.wrap_up(); //테스트, 리포트 후 청소
end
endprogram
