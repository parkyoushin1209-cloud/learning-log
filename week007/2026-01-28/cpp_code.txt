
template<class T>
char* as_bytes(T& i) {//T를 일련의 바이트로 취급
	void* addr = &i;// 객체를 저장하는 메모리 첫 바이트의 주소 가져오기
	return static_cast<char*>(addr); //그 메모리를 일련의 바이트로 취급
}


int main() {
	cout << "입력 파일 이름 입력\n";
	string iname;
	cin >> iname;
	ifstream ifs{ iname, ios_base::binary }; //스트림 모드 설정
	//binary로 열면 스트림이 바이트를 조작하지 않음
	if (!ifs)error("입력 파일 열기 실패");

	cout << "출력 파일 이름 입력\n";
	string oname;
	cin >> oname;
	ofstream ofs{ oname, ios_base::binary };
	//binary로 열면 스트림이 바이트를 조작하지 않음
	if (!ofs)error("출력 파일 열기 실패");

	vector <int> v;

	for (int x; ifs.read(as_bytes(x), sizeof(int));) // 바이트 읽기/ as_bytes로 메모리에서 바이트가 처음 시작되는 주소 반환, sizeof(int)로 바이트 크기 알려줌
		v.push_back(x);

	//.....v를 이용해 작업 수행

	//파일에 바이너리 쓰기
	for (int x : v)
		ofs.write(as_bytes(x), sizeof(int));
	return 0;
}
