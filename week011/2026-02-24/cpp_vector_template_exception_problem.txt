
template<typename T>
class SP : public S<T>{ // S가 아니라 S<T>로 작성
public:
	using S<T>::S;
	T& get();
	void set(T t) { S<T>::set(t); }
};

template<typename T>
T& SP<T>::get() {
	//return this->val; // dependent base class이므로 this->val로 작성
	return S<T>::get(); // private으로 바뀌면 기반 클래스의 함수 호출하는게 좋음
}


==============================
template<typename T>
struct S {
	S(T t)
		:val{t}{ }
	T& get() { return val; }
	const T& get() const { return val; }
	void operator=(const T&);
	S<T>& operator=(const S<T>&);
private:
	T val;
};

template<typename T>
void S<T>::operator=(const T& t) {
	val = t->val;
}

void set(T v) { val = v; }
set을 연산자 오버로딩으로 대체함
template<typename T> //올바른 복사 대입 연산자, 참조를 반환하는 것이 일반적
S<T>& S<T>::operator=(const S<T>& other) {
	val = other.val; // 멤버가 포인터나 외부 자원이면 얕은 복사 컨테이너 보통의 타입이면 깊은 복사
	return *this;
}

template<typename T>
void read_val(T& v) {
	cin >> v;
}
