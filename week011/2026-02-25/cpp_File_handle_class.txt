

class File_handle {
public:
    // 생성자
    explicit File_handle(const string& filename, const char* mode = "r")
        : file_(fopen(filename.c_str(), mode))  // .c_str c스타일 문자열로 반환 string에 있음 c함수는 std::string을 모르므로 사용
    {
        if (!file_) {
            throw runtime_error("파일을 열 수 없습니다.");
        }
    }

    // 복사 금지
    File_handle(const File_handle&) = delete;
    File_handle& operator=(const File_handle&) = delete;

    // 이동 허용
    File_handle(File_handle&& other) noexcept
        : file_(other.file_)
    {
        other.file_ = nullptr;
    }

    File_handle& operator=(File_handle&& other) noexcept { // noexcept는 예외를 던지지 않겠다는 것임
        if (this != &other) {
            close();
            file_ = other.file_;
            other.file_ = nullptr;
        }
        return *this;
    }

    // 소멸자
    ~File_handle() {
        close();
    }

    // 내부 FILE* 접근 필요하면 제공
    FILE* get() const { return file_; }

private:
    FILE* file_; // FILE은 stdio.h에 있는 구조체타입

    void close() {
        if (file_) {
            fclose(file_);// 포인터 반환
            file_ = nullptr;
        }
    }
};
