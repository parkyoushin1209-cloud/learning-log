interface X_if(input logic clk); // 인터페이스 정의
logic [7:0] din, dout;
logic reset_l, load;

clocking cb @(posedge clk); // reset_l은 비동기적 신호이므로 따로
output din, load; // 테스트벤치 기준으로 작성
input dout;  
endclocking

always @cb //monitor랑 다르게 always + strobe 조합을 사용해 해당 타임슬롯의 모든 할당이 끝나고 마지막에 출력
$strobe("@%0t : %m: out = %0d, in = %0d, ld = %0d, r = %0d",
        $time, dout, din, load, reset_l);

modport DUT(input clk, input din, load, reset_l,
            output dout);

modport TB(input load, clocking cb);
endinterface : X_if

module counter(X_if.DUT xi); //카운터
logic [7:0] count;
assign xi.dout = count;

always @(posedge xi.clk or negedge xi.reset_l) begin
    if(!xi.reset_l)  count <= '0;
    else if(xi.load) count <= xi.din;
    else             count <= count + 1;
end
endmodule

task reset();

xi.reset_l <= 1;
xi.cb.load <= 0;
xi.cb.din <= '0;
@(xi.cb) xi.reset_l <= 0;
@(xi.cb) xi.reset_l <= 1;

내용 정리
1. typedef로 타이핑 줄이고 모드포트를 잘 못 사용할 가능성을 줄일 수 있다.
2. 인터페이스 배열은 cross module refenence 방식과 포트를 이용한 방식으로 전달 가능하다.
3. 베릴로그에서 모듈에 파라미터화를 제공하듯 인터페이스에도 파라미터화를 적용 가능하다(여기서는 비트 폭 설정)
4. 인터페이스 내부에 절차적 코드(initial, always, routine, assertion 등)를 사용할 수 있다.
  
루틴의 내부 동작은 외부로 부터 숨겨져 있기에 구현을 미룰 수 있으며 모드 포트로 접근, 컨트롤 할 수 있다. 
또한 디자인, 테스트벤치에서 모두 사용 가능하다 이 방식의 장점은 버그의 공통된 출처 제거, 같은 프로토콜 사용 보장이다
다만 단점은 디자인, 테스트벤치가 코드를 공유해 설계, 검증 팀의 톡립성을 잃어버릴 수 있다는 점이다.
인터페이스 내부 어써션은 주로 protocol violation, unknown values를 체크하는 데 쓰인다.
기능적 커버리지는 커버리지 데이터 수집 트리거를 위해 이런 종류의 어써션을 사용한다.

5. 인터페이스 코드의 제한
  인터페이스 내부 태스크는 기능이 언격히 정의되는 RTL에서는 괞찮으나 verification IP 에서의 사용은 좋지 않다.
  인터페이스 와 이 안에 포함된 코드는 설정에 따라 확장되거나 동적으로 인스턴스화 될 수 없다.
  인터페이스는 private 데이터는 가질 수 없다.
  검증을 위한 어떤 코드라도 최대의 유연성, 설정 가능성을 필요로하므로 프로그램 블록 안에서 실행되는 클래스 안에 들어가야 한다.

6.결론
  sv에서 ifc는 생성하는 그룹 전체에 (연결성/타이밍/블록 사이 통신을 위한 기능) 등 강력한 기술을 제공한다.
  가상 ifc를 사용하면 신호 계층 코드는 런 타임때 가변 개수의 물리적 ifc에 연결될 수 있따.
  인테페이스는 신호를 구동하는 루틴, 프로토콜을 체크하는 어써션을 가질 수 있으나 **테스트는 프로그램 블록 안에 둬라**
  다방면으로 인터페이스는 포인터, 캡슐화, 추상화를 가진 클래스와 닮을 수 있으며 베릴로그의 포트, 넷보다 더 높은 수준에서 시스템을 모델링하는 ifc 생성을 허용한다.
