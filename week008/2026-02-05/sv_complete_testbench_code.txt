task Environment::run();
int num_gen_running;

// cpu ifc 가장 먼저 초기화
cpu.run();

num_gen_running = numRx;

// 각각의 Rx input channel에 대해 generator, driver 시작
foreach(gen[i]) begin
    int j = i; //automatic var holds index in spawned thread
    fork
        begin
            if(cfg.in_use_Rx[j])
            gen[j].run(); //wait for generator to finish
            num_gen_running--; //decrement driver count
        end
        if(cfg.in_use_Rx[j]) drv[j].run();
    join_none
end
//각각의 output Tx channel, 모니터 시작
foreach(mon[i]) begin
    int j = i; //  왜 automatic 안 쓰는지?---------------------------------------------------------
    fork
    mon[j].run();
    join_none
end
//모든 제너레이터 종료 또는 타임 아웃
fork : timeout_block
wait(num_gen_running == 0);
begin
    repeat(1_000_000) @(Rx[0].cbr);
    $display("@%0t: %m ERROR: Generator timeout ", $time);
    cfg.nErrors++;
end
join_any
disable timeout_block

// wait for the data to flow through switch, into monitors, scoreboard
repeat(1_000) @(Rx[0].cbr);
endtask : run

/*
랜덤 설정 헤더
nCells(시스템을 흐르는 셀의 전체 값을 위한 랜덤 값) 으로 시작
제약 c_nCells_valid는 유효 셀의 수가 0보다 큼을 보장
c_nCells_reasonable은 테스트를 합리적인 수로 제한(1000셀) 테스트에서 무효, 덮어쓰기 가능 

다이나믹 비트 배열 in_use_Rx 어떤 스위치로의 Rx 채널이 활성화됐는지 명시 run에 사용되어 활성화된 채널만 실행되게 함

배열 cells_per_chan 활성된 채널에 총 cell의 수를 랜덤하게 나눔
제약 zero_unused_channels는 비활성 채널에 대해 cell의 수를 0으로 설정
solver를 돕기 위해, 활성된 채널의 마스크는 채널 사이에 cell들을 나누기 전 먼저 해결
=> 아니면 cell의 수가 0으로 할당됐던 채널들만 비활성될 것임
*/

class Config;
int nErrors, nWarnings; //에러, 경고 수
bit [31:0] numRx, numTx; //copy of parameters

rand bit [31:0] nCells;  //total cells 0 < <1000
constraint c_nCells_valid
{  nCells > 0; }  
constraint c_nCells_reasonable
{  nCells < 1000; }

rand bit in_use_Rx[]; //input / output channel enabled
constraint c_in_use_valid
{   in_use_Rx.sum() > 0; } // 최소 1개

rand bit [31:0] cells_per_chan[];
constraint c_sum_ncells_sum // 전체 채널에 cell 나눔
{  cells_per_chan.sum() == nCells; }  // total number of cells

constraint zero_unused_channels  // 사용하지 않는 채널의 셀은 0으로 설정
{foreach(cells_per_chan[i]) 
{
    solve in_use_Rx[i] before cells_per_chan[i];
    if(in_use_Rx[i])
    cells_per_chan[i] inside {[1:nCells]};
    else cells_per_chan[i] == 0;
}
}

extern function new(input bit [31:0] numRx, numTx);
extern virtual function void display(input string prefix ="");
endclass : Config

class UNI_cell extends BaseTr; //constrained random ATM cell
rand bit         [3:0]  GFC;
rand bit         [7:0]  VPI;
rand bit         [15:0] VCI;
rand bit                CLP;
rand bit         [2:0]  PT;
rand bit         [7:0]  HEC;
rand bit  [0:47] [7:0]  Payload;

static bit syndrome_not_generated = 1;

extern function void post_randomize();
extern virtual function bit compare(input BaseTr to);
extern virtual function void display(input string prefix = "");
extern virtual function BaseTr copy(input BaseTr to = null);

extern function NNI_cell to_NNI();
extern function void generate_syndrome();
extern function bit [7:0] hec (bit [31:0] hdr);

function UNI_cell::new();
if(syndrome_not_generated)
generate_syndrome();
endfunction

function void UNI_cell::post_randomize();
HEC = hec({GFC, VPI, VCI, CLP, PT});
endfunction

function bit UNI_cell::compare(input BaseTr to);
UNI_cell c;
$cast(c, to);
if (this.GFC != c.GFC)           return 0;
if (this.VPI != c.VPI)           return 0;
if (this.VCI != c.VCI)           return 0;
if (this.CLP != c.CLP)           return 0;
if (this.PT != c.PT)             return 0;
if (this.HEC != c.HEC)           return 0;
if (this.Payload != c.Payload)   return 0;
return 1;
endfunction

function void UNI_cell::display(input string prefix = "");
ATMCellType p;

$display("%sUNI id:%0d GFC=%x, VPI=%x, VCI=%x, CLP=%b, PT=%x, HEC=%x, Payload[0]=%x",
prefix, id, GFC, VPI, VCI, CLP, PT, HEC, Payload[0]);
this.pack(p);
$write("%s",prefix);
foreach(p.Mem[i]) $write("%x ",p.Mem[i]);
$display;
endfunction

else            $cast(copy, to);
copy.GFC         = this.GFC;
copy.VPI         = this.VPI; 
copy.VCI         = this.VCI; 
copy.CLP         = this.CLP; 
copy.PT          = this.PT;
copy.HEC         = this.HEC;
copy.Payload     = this.Payload;
return copy;
endfunction

this.CLP           = from.uni.CLP;

function NNI_cell UNI_cell::to_NNI(); // 유니셀로부터 NNI 셀 생성 스코어보드에 씀

copy = new();
copy.VPI         = this.VPI; 
copy.VCI         = this.VCI; 
copy.CLP         = this.CLP; 
copy.PT          = this.PT;
copy.HEC         = this.HEC;

function void UNI_cell::generate_syndrome(); // HEC 계산에 이용되는 신드롬 배열 생성
bit [7:0] sndrm;
for (int i = 0; i < 256 ; i = i + 1) begin
    sndrm = i;
    reapeat(8) begin
        if(sndrm[7] === 1'b1)
        sndrm = (sndrm << 1) ^ 8'h07;
        else
        sndrm = sndrm << 1;
    end
    syndrome[i] = sndrm;
end
syndrome_not_generated = 0;
endfunction

function bit [7:0] UNI_cell::hec(bit [31:0] hdr); // 이 객체를 위한 HEC 값 계산
hec = 8'h00;
repeat(4) begin
    hec = syndrome[hec ^ hdr[31:24]];
    hdr = hdr << 8;
end
hec = hec ^ 8'h55;
endfunction

endclass

function Driver::new(input mailbox gen2drv,
                    input event drv2gen,
                    input vUtopiaRx Rx,
                    input int PortID);
this.gen2drv = gen2drv;
this.drv2gen = drv2gen;
this.Rx = Rx;
this.PortID = PortID;
endfunction

task Driver::run(); // 발생기에서 트랜잭션 받아서 DUT로
UNI_cell c;
bit drop = 0;

//포트 초기화
Rx.cbr.data <= 0;
Rx.cbr.soc  <= 0;
Rx.cbr.clav <= 0;

forever begin
    gen2drv.peek(c); //메일 박스의 앞에서 셀 읽기
    begin: Tx  //전 전송 콜백
    foreach(cbsq[i]) begin
        cbsq[i].pre_tx(this, c, drop);
        if(drop) disable Tx; // 이 셀 보내지 마
    end

    c.display($sformat("@%0t: Drv%0d: ", $time, PortID));
    send(c);
    foreach(cbsq[i])// 후 전송 콜백
    cbsq[i].post_tx(this, c);
    end : Tx

    gen2drv.get(c);
    ->drv2gen; //uni_generator에게 끝났다고 알림
end
endtask

task Driver::send(input UNI_cell c);
ATMCellType Pkt;

c.pack(Pkt);
$write("Sending cell: ");
foreach(Pkt.Mem[i])
$write("%x ", Pkt.Mem[i]); $display;

@(Rx.cbr);
Rx.cbr.clav <= 1;
for(int i=0; i <= 52; i++)begin
    while (Rx.cbr.en === 1'b1) @(Rx.cbr); // enable 아니면 루프
    //Assert start of cell, assert enable, send byte 0 (i==0)
    Rx.cbr.soc <= (i == 0);
    Rx.cbr.data <= Pkt.Mem[i];
    @(Rx.cbr);
end
Rx.cbr.soc <= 'z;
Rx.cbr.data <= 8'bx;
Rx.cbr.clav <= 0;
endtask

save_expected는 콜백 Sdb_Driver_cbs의 post_tx에서 호출됨

class Expect_cells;
NNI_cell q[$];
int iexpect, iactual;
endclass

class Scoreboard;
Config cfg;
Expect_cells expect_cells[];
NNI_cell cellq[$];
int iexpect, iactual;

extern function new(Config cfg);
extern virtual function void wrap_up();
extern function void save_expectes(UNI_cell ucell);
extern function void check_actual (input NNI_cell c,
                                   input int portn);
extern function void display(string prefix="");
endclass

function Scoreboard::new(Config cfg);
this.cfg = cfg;
expect_cells = new[NumTx]; // 여기에는 왜 num 말고 Num 쓰는지?
foreach(expect_cells[i])
expect_cells[i] = new();
endfunction


function void Scoreboard::save_expectes(UNI_cell ucell);
NNI_cell ncell = ucell.to_NNI;
CellCfgType CellCfg = top.squat.lut.read(ncell.VPI);

$display("@%0t: Scb save: VPI=%0x, Forward=%b",
           $time, ncell.VPI, CellCfg.FWD);
ncell.display($sformat("@%0t: Scb save: ", $time));

for(int i=0; i <NumTx; i++) // 이 셀이 보내질 모든 Tx ports 위치 찾기
if(CellCfg.FWD[i]) begin
    expect_cells[i].q.push_back(ncell);
    expect_cells[i].iexpect++;
    iexpect++;
end
endfunction

function void Scoreboard::check_actual (input NNI_cell c,
                                        input int portn);
NNI_cell match;
int match_idx;

c.display($sformat("@%0t: Scb check: ", $time));

if(expect_cells[portn].q.size() == 0) begin
    $display("@%0t: ERROR: %m cell not found, SCB TX%0d empty", $time, portn);
    c.disply("Not Found: ");
    cfg.nErrors++;
    return;
end

expect_cells[portn].iactual++;
iactual++;

foreach(expect_cells[portn].q[i]) begin
if(expect_cells[portn].q[i].compare(c)) begin
    $display("@%0t: Match found for cell", $time);
    expect_cells[portn].q.delete(i);
    return
end
end

$display("@%0t: ERROR: %m cell not found", $time);
c.display("Not Found: ");
cfg.nErrors++;
endfunction

function void Scoreboard::wrap_up();
//print end of simulation report
$display("@%0t: %m %0d expected cells, %0d actual cells rcvd", $time, iexpect, iactual);

//Look for leftover cells
foreach(expect_cells[i]) begin
    if(expect_cells[i].q.size()) begin
        $display("@%0t: %m cells in SCB Tx[%0d] at end of test", $time, i);
        this.display("Unclaimed: ");
        cfg.nErrors++;
    end
end
endfunction

//print the contents of the scoreboard, mainly for debugging
function void Scoreboard::display(string prefix="");
$display("@%0t: %m so far %0d expected cells, %0d actual rcvd", $time, iexpect, iactual);

foreach(expect_cells[i]) begin
    $display("Tx[%0d]: exp=%0d, act=%0d", i, expected_cells[i].iexpect, expect_cells[i].iactual);
foreach(expected_cells[i].q[j])
expected_cells[i].q[j].display($sformat("%sScoreboard: Tx%0d: ", prefix, i));
end
endfunction

class CPU_driver;

Config cfg;
bit [NumTx-1:0] fwd;

extern function new(vCPU_T mif, Config cfg);
extern task Initialize_Host();
extern task HostWrite(int a, CellCfgType d);
extern task HostRead (int a, output CellCfgType d);
extern task run();
endclass

function CPU_driver::new(vCPU_T mif, Config cfg);
this.mif = mif;
this.cfg = cfg;
endfunction

task CPU_driver::Initialize_Host();
mif.BusMode <= 1;
mif.Addr <= 0;
mif.DataIn <= 0;
min.Sel <= 1;
mif.Rd_DS <= 1;
mif.Wr_RW <= 1;
endtask

task CPU_driver::HostWrite(int a, CellCfgType d);
#10 mif.Addr <= a; mif.DataIn <= d; mif.Sel <= 0;
#10 mif.Wr_RW <= 0;
while(mif.Rdy_Dtack !== 0) #10;
#10 mif.Wr_RW <= 1; mif.Sel <= 1;
while(mif.Rdy_Dtack==0) #10;
endtask

task CPU_driver::HostRead (int a, output CellCfgType d);
#10 mif.Addr <= a; mif.Sel <= 0;
#10 mif.Rd_DS <= 0;
while (mif.Rdy_Dtack!==0) #10;
#10 d = mif.DataOut; mif.Rd_DS <= 1; mif.Sel <= 1;
while (mif.Rdy_Dtack==0) #10;
endtask

task CPU_driver::run();
CellCfgType CellFwd;
Initialize_Host();
//configure through Host interface
repeat(10) @(negdege clk);
$write("Memory: Loading ... ");
for(int i=0;i<=255;i++) begin
    CellFwd.FWD = $urandom();
`ifdef FWDALL
CellFwd.FWD = '1
`endif 
CellFwd.VPI = i;
HostWrite(i, CellFwd);
lookup[i] = CellFwd;
end

//verify memory
$write("Veriying ...");
for(int i=0; i <=255; i++) begin
    HostRead(i, CellFwd);
    if(lookup[i] != CellFwd) begin
        $display("FATAL, Mem Loc 0x%x contains 0x%x, expected 0x%x",
                   i, CellFwd, lookup[i]);
                   $finish;
    end
end
$display("Verified");

endtask
