`define SV_RAND_CHECK(r)\ //매크로 정의 randomize() 실패시 에러 문자
do begin\
if(!(r)) begin
  $display("%s:%0d:Randomization failed\"%s\"",\
  `_FILE_, `_LINE_, `"r`");\
  $finish;\
end\
end while(0)

class Transaction;   // 값, 가중치 부여
rand bit[1:0] src, dst;
constraint c_dist{
  src dist {0:=40, [1:3] := 60}; // 0 : 40/220 1: 60/220 2: 60/220 3: 60/220
  dst dist {0:=40, [1:3] /= 60}; // 0 : 40/100 1: 20/100 2: 20/100 3: 20/100
}
endclass

class BusOp;
typedef enum {BYTE, WORD, LWRD} length_e;
rand length_e len;
bit [31:0] w_byte = 1, w_word = 3, w_lwrd = 5;
constraint c_len{
  len dist {BYTE := w_byte,  // 1/9
            WORD := w_word,  // 3/9
            LWRD := w_lwrd}; // 5/9
}
endclass

class Ranges;
rand bit [31:0] c;
bit [31:0] lo, hi;
constraint c_range{
  c inside {[lo:hi]}; // lo <= c <= hi 모든 확률은 동일 !(c inside {[lo:hi]}) c < lo && hi < c
}
endclass

class Fib;
rand bit [7:0] f;
bit [7:0] vals[] = '{1,2,3,5,8};
constraint c_fibonacci{

class RandcInside;

int array[];

function pick();

endfunction

endclass

class SolveXBeforeY;
rand bit x;
rand bit[1:0] y;
constraint c_xy{
  (x==0) -> y==0; //x가 0이면 y도 0 ==(x==1||y==0) x가 1이거나 y가 0이거나 혹은 둘다 거나
  solve x before y;
}
endclass
