class Transaction;
bit [31:0] addr, csm, data[8];

function new(input logic [31:0] a =3, d = 5);
addr = a;
data = '{default : d}; // csm은 기본값으로 x를 가짐 2-state면 0
endfunction

function void display();
$display("Transaction: %h", addr);
endfunction: display

function void calc_csm();
csm = addr ^ data.xor;
endfunction: calc_sum
endclass : Transaction

initial begin
  Transaction tr;
  tr = new(.a(10)); // a는 10, d는 5
end
