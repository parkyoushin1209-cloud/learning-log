void skip_to_int() // int를 찾을면 버퍼에 정수를 되돌려놓음, 유틸리티 함수
{
	if (cin.fail()) { // 다른 타입의 입력을 받을 경우
		cin.clear();

		for (char ch; cin >> ch;) { //문자가 입력된 경우 계속 실행
			if (isdigit(ch) || ch == '-') { //숫자나 -면 버퍼에 되돌리고 끝냄
				cin.unget();
				return;
			}
		}
	}
}

int get_int() {
	int n = 0;
	while (true) {
		if (cin >> n)return n;

	}

  operator>>()에서 사용하지 않은 문자는 소모하지 않아야 함
  istream& operator>>(istream& is, Date& dd) {
	int y, m, d;
	char ch1, ch2, ch3, ch4;
	is >> ch1 >> y >> ch2 >> m >> ch3 >> d >> ch4;
	if (!is) return is;
	if (ch1 != '(' || ch2 != ',' || ch3 != ',' || ch4 != ')') { 
		is.clear(ios_base::failbit); //입력 형식에 맞지 않으면 fail() 상태로 설정
		return is;
	}
	dd = Date{ y, Date::Month(m), d };//(정수,정수,정수) 형식으로 받아야 dd 수정
	return is;
}
