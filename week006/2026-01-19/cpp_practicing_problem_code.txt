template<class T, class U>
T narrow_cast(U u)
{
	T t = static_cast<T>(u);
	if (static_cast<U>(t) != u)
		error("narrow_cast error");
	return t;
}

template<class T>
void print_vector(const vector<T>& v, const string& label = "")
{
	if (!label.empty()) cout << label << ": ";
	for (const auto& x : v) cout << x << ' ';
	cout << '\n';
}

void merge_s(vector <string>& v, int left, int mid, int right) {  //이미 정렬된 두 배열을 하나로 합침, 작은 값부터 temp에 넣음
	int i = left;
	int j = mid + 1;
	int k = 0;

	vector <string> temp(right - left + 1);
	while (i <= mid && j <= right) {  // 두 부분 배열을 비교하며 병합
		if (v[i] <= v[j]) {
			temp[k++] = v[i++];
		}
		else {
			temp[k++] = v[j++];
		}
	}
	//두 whlie문은 순서 상관 없음 -> 어차피 한쪽 배열에만 값이 남아 있기 때문에
	while (i <= mid)  //왼쪽 배열에 남은 것들
		temp[k++] = v[i++];

	while (j <= right) // 오른쪽 배열 남은 것들  
		temp[k++] = v[j++];


	for (i = left, k = 0; i <= right; i++, k++)  	//원래 배열로 복사
		v[i] = temp[k];
}



void mergesort_s(vector <string>& v, int left, int right) {   //배열을 반으로 계속 분할, 끝까지 가면 멈춤 left >= right
	if (left < right) { //단일 요소에서는 병합 x (if문 실행 안됨)
		int mid = (left + right) / 2;

		mergesort_s(v, left, mid);
		mergesort_s(v, mid + 1, right);
		merge_s(v, left, mid, right);
	}
}

bool Book::is_valid_isbn(const string& isbn) const {
	int cnt_li = 0;
	int cnt_ch = 0;
	for (char c : isbn) {
		switch (c) {
		case 'n': case 'x':
			cnt_li++;
			break;
		case '-':
			cnt_ch++;
			break;
		}
	}
	if (cnt_li >= 4 && cnt_ch >= 3)
		return true;
	else
		return false;
}

bool operator!=(const Book& book1, const Book& book2) {
	return !(book1 == book2);
}

ostream& operator<<(ostream& os, const Book& book) {
	cout << book.title << "\n"
		<< book.author_name << "\n"
		<< book.ISBN << "\n";
}

bool Book::is_valid() const {
	return is_valid_isbn(ISBN) 
		&& !title.empty()
		&& !author_name.empty();
}


bool operator==(const Book& book1, const Book& book2)
{
	return (book1.Isbn() == book2.Isbn())
		&& (book1.Title() == book2.Title())
		&& (book1.Author_name() == book2.Author_name())
		&& (book1.Copyright_information() == book2.Copyright_information());
}
