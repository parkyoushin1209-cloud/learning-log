/* 시뮬레이션 커맨드 라인
vsim testbench -coverage +UVM_TESTNAME=rand_test
vsim +UVM_TESTNAME=add_test top 형식으로 작성
*/

module top;
import uvm_pkg::*; // 보통 패키지, 매크로를 함깨 가져오는 방식을 자주 사용
`include "uvm_macros.svh"

import tinyalu_pkg::*;
`include "tinyalu_macros.svh"

tinyalu_bfm bfm();
tinyalu DUT(.A(bfm.A), .B(bfm.B), .clk(bfm.clk), 
            .op(bfm.op), .reset_n(bfm.reset_n),  
            .start(bfm.start), .done(bfm.done), .result(fm.result));

initial begin
    uvm_config_db #(virtual interface tinyalu_bfm) :: set(null, "*", "bfm", bfm);
    //uvm은 생성자에서 구체적 인자를 요구하므로 객체의 생성자에 bfm 핸들 전달이 불가능
    //대신 uvm_config_db를 사용 -> 테스트벤치 전체에서 정렬된 방식으로 전역 변수 저장을 쉽게 하기 위해 static class, parameterization 이용함
    run_test();
    // 전역 장소에 BFM을 한 번 저장했으면 테스트 시작 가능
    //uvm_pkg에 저장돼있으며 호출해서 테스트 시작 가능
    // 시뮬레이터 커맨드 라인에서 +UVM_TESTNAME 뒤 문자열을 읽어와 팩토리를 이용해 해당 클래스명의 객체 인스턴스화
    // run_test에 직접 테스트 클래스명을 전달하면 +UVM_TESTNAME 사용을 피할 수 있으나 한 번의 컴파일로 여러 테스트를 실행하는 이점이 사라짐
end
endmodule : top

class random_test extends uvm_test; // random_test는 uvm_test에서 확장 uvm_test는 uvm_component에서 확장됨

`uvm_component_utils(random_test);
// 이 매크로는 클래스문 바로 다음에 두기, random_test 클래스를 팩토리에 저장하고 이 후 이 클래스의 객체 생성 가능함
virtual interface tinyalu_bfm bfm;

function new(string name, uvm_component parent); 
// uvm_component의 확장을 위해 생성자의 인자로 string 타입 name, uvm_component 타입 parent를 순서대로 가져야 하며
super.new(name, parent); // 첫 줄은 super.new(name, parent)가 되어야 함 이 호출이 없다면 uvm이 적절히 동작하지 않을 것임
if(!uvm_config_db #(virtual interface tinyalu_bfm)::get(null, "*", "bfm", bfm))
// uvm_config_db의 get함수를 사용해 bfm의 핸들을 얻으며 이때 set에서 사용한 문자열 "bfm"과 BFM을 저장할 변수를 함께 전달함
// get은 BFM 얻지 못했다면 0을 반환함 
$fatal("Failed to get BFM");
endfunction

//top 모듈에서 run_test() 호출 시 uvm은 팩토리를 사용해 테스트 객체를 생성하고 run_phase 메서드를 호출해 시작시킨다.
task run_phase(uvm_phase phase); 
// UVM은 uvm_test가 run_phase()를 가지게 정의하고 있고 test 실행을 위해 run_phase()를 이용한다.
//run_phase 메서드를 반드시 오버라이드 해야 하며 phase는 uvm_phase 타입의 phase라 불리는 단일 인자를 가져야한다.
random_tester random_tester_h;
coverage coverage_h;
scoreboard scoreboard_h;

phase.raise_objection(this);
random_tester_h = new(bfm);
coverage_h = new(bfm);
scoreboard_h = new(bfm);

fork
    coverage_h.execute();
    scoreboard_h.execute();
join_none

random_tester_h.excute();
phase.drop_objection(this); // run_objection과 함께 이용함으로써 테스트벤치가 언제 멈출 수 있는지 알려줌
endtask : run_phase

endclass


class add_test extends uvm_test;
`uvm_component_utils(add_test);

virtual interface tinyalu_bfm bfm;

function new(string name, uvm_component parent);
super.new(name, parent);
if(!uvm_config_db #(virtual interface tinyalu_bfm) :: get(null, "*", "bfm", bfm))
$fatal("Failed to get BFM");
endfunction 

task run_phase(uvm_phase phase);
add tester add_tester_h; \
