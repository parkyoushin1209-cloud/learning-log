package tinyalu_pkg;
typedef enum bit[2:0] {no_op = 3'b000,
                       add_op = 3'b01,
                       and_op = 3'b010,
                       xor_op = 3'b011,
                       mul_op = 3'b100,
                       rst_op = 3'b111} operation_t;

endpackage

interface tinyalu_bfm; // => tinyalu TB의 모든 신호 캡슐화, 클럭, reset_alu() task, send_up task{alu로 op 보냄} 제공
import tinyalu_pkg::*;

operation_t op_set;
byte unsigned A; // 패키지로 들어가면 ifc.A 형식으로 접근 불가능
byte unsigned B;
bit clk;
bit reset_n;
wire [2:0]wire;
bit state;
wire done;
wire [15:0] reslt;

task send_op(input byte iA, input byte iB, input operation_t iop, 
             output shortint ale_reset);
// 연산자 및 데이터 취하고 start 신호를 올린 후 요청된 연산자에 근거해 다시 내림 
// => 이런 캡슐화는 두 가지 이득이 있음
//1. 프로토콜 수준 동작을 코드 여기저기 넣어둘 필요 없음(신호들을 다루는 코드보다 이 태스크 호출이 더 간편)
//2. 하나의 장소에 모든 프로토콜 동작 수정 가능(여기에서의 수정 사항은 설계 전체에 퍼짐)
//operation_t는 패키지에 정의된 열거형 타입임 
op_set = iop;

if(iop == rst_op) begin
    @(posedge clk);

end else begin
    @(negedge clk);
    A=iA;
    B=iB;
    start = 1'b1;
    if(iop == no_op) begin
        @(posedge clk);
        #1;
        start = 1'b0;
    end else begin
        do
        @(negedge clk);
        while(done == 0);
        start = 1'b0;
    end
end
endtask

task reset_alu();
reset_n = 1'b0;
@(negedge clk);
@(negedge clk);
reset_n = 1'b1;
start = 1'b0;
endtask : reset_alu

endinterface

class scoreboard;
virtual tinyalu_bfm bfm;

function new(virtual tinyalu_bfm bfm);
this.bfm = bfm;
endfunction

task execute();
shortint predicted_result;
forever begin : self_checker
@(posedge bfm.done)
case(bfm.op_set)
add_op : predicted_result = bfm.A + bfm.B;
and_op : predicted_result = bfm.A & bfm.B;
xor_op : predicted_result = bfm.A ^ bfm.B;
mul_op : predicted_result = bfm.A * bfm.B;
endcase

if((bfm.op_set != no_op) && (bf,.op_set != rst_op))
if(predicted_result != bfm.result)
$error("FAILED: A: %0h  B: %0h  op: %s  result: %0h",
bfm.A, bfm.B, bfm.op_set.name(), bfm.result);
end
endtask

endclass

class Covgroup;
virtual tinyalu_bfm bfm;

covergroup op_cov; 
coverpoint op_set {
    bins single_cycle[] = {[add_op : xor_op], rst_op, no_op};
    bins multi_cycle = {mul_op};
}

endgroup

covergroup zeros_or_ones_on_ops; 

all_ops : coverpoint op_set{ // rst, no를 제외한 모든 연산자가 사용됐는가?
    ignore_bins null_ops = {rst_op, no_op};
}

a_leg : coverpoint A{ // 모든 가능한 값들이 테스트 됐는가?
    bins zeros = {'h00};
    bins others = {['h01 : 'hFE]};
    bins ones = {'hFF};
}

endgroup


function new(virtual tinyalu_bfm bfm);
this.bfm = bfm;
op_cov = new();
zeros_or_ones_on_ops = new();
endfunction
