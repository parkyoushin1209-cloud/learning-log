/* 레이어드 테스트벤치 구조 요구
레이어	역할
Interface / Signals	ALU 입력/출력 시그널 정의
Driver Layer	a, b, op 값을 생성해서 ALU에 적용
Checker Layer	ALU 출력 y가 예상값과 맞는지 확인
Test Sequence Layer	다양한 연산 케이스 순서대로 실행
00=ADD, 01=SUB, 10=AND, 11=OR
테스트 케이스

a	b	op	예상 y
5	3	ADD	8
10	4	SUB	6
6	3	AND	2
6	3	OR	7
*/

`define SV_RAND_CHECK(r)\
do begin\
if(!r) begin
    $display("%s:%0d:Randomization failed"\%s\"",\
    `__FILE__,`__LINE__,`"r`");\
    $finish;\
end\
end while(0)

`timescale 1ns/1ps

interface ifc_bus(input bit clk);

logic [7:0] a, b, y;
logic [1:0] op;
bit rst;

clocking cbr @(posedge clk);
input Trans_type cell; // BaseTr이 input이면 서브클래스도 신호로 넣을 수 있음
endclocking
modport Rx(clocking cbr);

clocking cbt @(posedge clk);
output Trans_type cell;
endclocking
modport Tx(clocking cbt);

modport DUT(input a, b, op, clk, rst,
            output y);

modport TEST(output a, b, op, 
             input rst, y);

endinterface

module alu_8bit (ifc_bus.DUT bus);

    always @(posedge bus.cb.clk) begin
    if(bus.rst) bus.y <= 0;
    else begin
       case(bus.op)
            2'b00: bus.y <= bus.a + bus.b;
            2'b01: bus.y <= bus.a - bus.b;
            2'b10: bus.y <= bus.a & bus.b;
            2'b11: bus.y <= bus.a | bus.b;
        endcase
    end
    end
endmodule

typedef virtual ifc_bus.TEST Vbus_tb;
typedef virtual ifc_bus.Rx Vbus_Rx;
typedef virtual ifc_bus.Tx Vbus_Tx;

module top();
bit clk,rst;
initial begin
clk = 0;
forever #5ns clk = ~clk;
end

initial begin
    rst = 0;
    #10 rst = 1;
end

ifc_bus ifc(clk);
alu_8bit alu(ifc.DUT bus);
Test test(ifc.TEST);
endmodule


program automatic Test(ifc_bus.TEST bus);
Environment env;

initial begin
    env = new(bus);
    env.gen_cfg();
    env.build();
    env.run();
    env.wrap_up();
end
endprogram




class Environment; // 환경 생성자에서는 메일 박스를 따로 받을 필요가 없음
Driver drv;
Generator gen;
Scoreboard scb;
Monitor mon;
mailbox gen2drv;
event drv2gen;
virtual fc_bus ifc;
Config cfg;
Coverage cov;

extern function void new(input ifc_bus ifc);
extern virtual function void gen_cfg();
extern virtual function void build();
extern virtual task run();
extern virtual function void wrap_up();
endclass

function void Environment::new(input ifc_bus ifc);
this.ifc = ifc;
cfg = new();
endfunction

function void Environment::gen_cfg();
`SV_RAND_CHECK(cfg.randomize());
endfunction

function void Environment::build();
gen2drv = new(); //drv, gen보다 먼저 생성해주기
drv = new(gen2drv, drv2gen); // 여기서 메일박스, 이벤트 넣어주기 
gen = new(gen2drv, drv2gen);
scb = new(cfg); // cfg 넣기
mon = new();
cov = new();

begin  //콜백 등록은 여기서
    Scb_drv_cbs sdc = new(scb);
    Scb_mon_cbs smc = new(scb);
    drv.cbsq.push_back(sdc);
    mon.cbsq.push_back(smc);
end

begin  //initial은 붙이면 안 됨
    Cov_mon_cbs cmc = new(cov);
    mon.cbsq.push_back(cmc);
end

endfunction

task Environment::run();
fork 
drv.run();
gen.run();
mon.run();
scb.run();
cov.run();
join
endtask

function void Environment::wrap_up();
$display("@%0t: End of sim, %0d errors, %0d warnings", $time, cfg.nErrors, cfg.nWarnings);
scb.wrap_up();
endfunction

class Config;
int nErrors, nWarnings;
rand bit [31:0] nTrans;

constraint nTrans_valid{
    0 < nTrans;
    nTrans < 1000; 
}

endclass

typedef struct packed{
bit[7:0] a, b;
bit[1:0] op;
} trantype; //18바이트

typedef union packed{
trantype Tr;
bit[17:0] Mem; //[0:8][1:0]이 [8:0][1:0]보다 인덱스 방향이 명황해 읽기 쉬움 0 1 2 3...[1:2](전자)  8 7 6 5....[1:2](후자)   
} Trans_type;

class BaseTr; // 빈 상태로 두어 구현 의존성을 없애 다형성을 지킴
endclass

class Transaction extends BaseTr;
rand bit[7:0] a,b;
rand bit[1:0] op;

virtual function BaseTr copy(input BaseTr to = null); // BaseTr의 입력에 null 넣고 상황에 따라 new() or cast
if(to == null) copy.new();
else $cast(copy, to);
copy.a = this.a;
copy.b = this.b;
copy.op = this.op;
return copy;


endfunction

endclass

class Generator;
mailbox gen2drv;
event drv2gen;
int nTrans; // 여기에 나옴 => 트랜잭션을 nTrans번 반복할 수 있도록
BaseTr blueprint;

function void new(input mailbox mbx, input event ev, input int nTrans);
this.gen2drv = mbx;
this.drv2gen = ev;
this.nTrans = nTrans;
blueprint = new();
endfunction

virtual task run();
repeat(nTrans) begin
BaseTr tr;
`SV_RAND_CHECK(blueprint.randomize());
$cast(tr, blueprint.copy());
gen2drv.put(tr);
@drv2gen;
end
endtask

endclass // generator에 wrap_up 없음
