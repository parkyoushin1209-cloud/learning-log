void fib(svLogicVecVal data[20]){
    data[0].aval = 1;
    data[1].aval = 1;
    data[0].bval = 0;
    data[1].bval = 0;
    for(int i=2; i<20;i++){
      data[i].aval = data[i-1].aval + data[i-2].aval; // aval은 비트 벡터이며 실질적인 값을 나타냄 aval[0]는 상태를 나타내시도 하며 값을 나타내기도 함
      data[i].bval = 0; //bval clear
     }
}

  void fib_oa(const svOpenArrayHandle data_oa){ //svOpenArrayHandle은 open array와 관련된 정보를 담고 있는 구조체를 가리키는 핸들임
    int i, *data;
    data = (int*) svGetArrayPtr(data_oa); //배열의 첫 요소 주소 반환 및 int*로 접근(void* 그대로는 산술 접근 불가능)
    data[0] = 1;
    data[1] = 1;
    for(i = 2; i<=svSize(data_oa, 1); i++) // 1은 차원을 나타내며 data_oa 사이즈를 반환
    data[i] = data[i-1] + data[i-2];
}

void mydisplay(const svOpenArrayHandle h){
    int i, j;
    int lo1 = svLow(h, 1);
    int hi1 = svHigh(h,1);
    int lo2 = svLow(h, 2);
    int hi2 = svHigh(h, 2);
    for(i = lo1; i<=hi1 ; i++){
        for( j = lo2 ; j<=hi2; j++){
          int *a = svGetArrElemPtr2(h, i, j); // 포인터를 반환
            io_printf("C: a[%d][%d] = %0d", i, j, *a);
            *a = i*j;
        }
    }
}
