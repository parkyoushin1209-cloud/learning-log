  class Counter7{ //Counter7 class 정의
    public:
    Counter7();
    void count();
    void load(const svBitVecVal* i);
    void reset();
    int get();
    private:
    unsigned char cnt;
};

Counter7::Counter7{
    cnt = 0;
}

void Counter7::count(){
    cnt = cnt + 1;
}

void Counter7::load(const svBitVecVal* i){
    cnt = *i;
    cnt &= 0x7f;
}

void Counter7::reset(){
    cnt = 0;
}

int Counter7::get(){
return cnt;
}

chandle inst;

inst = Counter7_new(); // 이거만 가져온 형식 따르면 됨

function void count();
Counter7_count(inst);
endfunction

function void load(input bit[6:0] val);
Counter7_load(inst, val);
endfunction

function void reset();
Counter7_reset(inst);
endfunction

  function bit[6:0] get();     //  int형을 반환함 bit[6:0]은 svBitVecVal을 가리키는 포인터를 반환해야 할 수도 있으나
    return Counter7_get(inst); // 포인터 반환은 불가능하며 다음의 타입들만 반환 가능 => c의 기본 스칼라 범위와 일치하지 않음 bit[6:0]은 7비트임
